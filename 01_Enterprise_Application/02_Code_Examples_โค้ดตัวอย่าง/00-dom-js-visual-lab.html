เพื่อดูว่าโค้ด JavaScript บรรทัดไหนเป็นคนสั่งเปลี่ยนค่าในกล่องนั้น

          

        


        
        

          
📝 โค้ด JavaScript จริงที่ใช้ควบคุมหน้าจอนี้ (คลิกแต่ละบรรทัดได้)

          

            
function render(data) {

            
  const percent = (data.actual / data.target) * 100;

            
  const status = getStatus(percent); // คืนค่า { key: 'ok'|'warn'|'bad', text: '...' }

            

            
  document.getElementById('line').textContent = `Line ${data.line} · กะ ${data.shift}`;

            
  document.getElementById('actual').textContent = data.actual.toLocaleString();

            
  document.getElementById('target').textContent = data.target.toLocaleString();

            
  document.getElementById('defect').textContent = data.defect;

            
  document.getElementById('performance').textContent = percent.toFixed(1) + '%';

            

            
  const statusEl = document.getElementById('status');

            
  statusEl.textContent = status.text;          // เปลี่ยนคำเป็น ปกติ / เฝ้าระวัง / ต่ำกว่าเป้า

            
  statusEl.className = 'status ' + status.key; // เปลี่ยนสีป้ายตามคลาส .ok / .warn / .bad

            

            
  // เปิดปุ่มให้กดได้เฉพาะตอนสถานะเป็น 'bad' (ต่ำกว่าเป้า)

            
  document.getElementById('alert-btn').disabled = (status.key !== 'bad');

            
}

          


          

            🔗 เชื่อมต่อไปยังเรื่อง API (Workshop 3 สไลด์หน้า 84–85):

            พอเราเขียนฟังก์ชัน render(data) ที่มี DOM ครบแบบนี้แล้ว ตอนดึงข้อมูลจริงจาก Server ด้วย const data = await res.json(); เราก็แค่เรียก render(data); บรรทัดเดียว หน้าเว็บทั้งหมดก็อัปเดตตัวเลขเองอัตโนมัติทันที!
